# 视频下载使用示例

## YouTube 下载示例

# 下载单个视频（最佳质量）
yt-dlp "https://www.youtube.com/watch?v=VIDEO_ID"

# 下载 720p 视频
yt-dlp -f "bestvideo[height<=720]+bestaudio/best[height<=720]" "YouTube URL"

# 下载整个播放列表
yt-dlp "https://www.youtube.com/playlist?list=PLAYLIST_ID"

# 只下载前 10 个视频
yt-dlp --playlist-end 10 "播放列表URL"

# 下载字幕（中文）
yt-dlp --write-subs --sub-langs zh-Hans,zh-Hant "YouTube URL"

# 只下载音频（MP3）
yt-dlp -x --audio-format mp3 "YouTube URL"

# 下载直播
yt-dlp "https://www.youtube.com/live/LIVE_STREAM_ID"


## B站下载示例

# 下载单个视频
yt-dlp "https://www.bilibili.com/video/BV..."

# 下载最高画质（需要 cookies）
yt-dlp --cookies-browser chrome "B站视频URL"

# 下载系列视频
yt-dlp "https://www.bilibili.com/medialist/detail/..."

# 下载弹幕（需要额外工具）
# yt-dlp --write-subs "B站视频URL"


## Twitter/X 下载示例

# 下载推文中的视频
yt-dlp "https://twitter.com/user/status/TWEET_ID"

# 下载高质量版本
yt-dlp -f "best" "推文URL"

# 批量下载用户的所有视频
yt-dlp "https://twitter.com/USERNAME"


## 抖音下载示例

# 下载抖音视频
yt-dlp "抖音分享链接"

# 去除水印（如果支持）
yt-dlp --format "best" "抖音分享链接"


## 批量下载示例

# 从文件读取 URL 列表
# 创建 urls.txt 文件，每行一个 URL
yt-dlp -a urls.txt

# 下载播放列表并创建归档
yt-dlp --download-archive archive.txt "播放列表URL"


## 高级用法示例

# 自定义输出路径和文件名
yt-dlp -o "~/Downloads/%(title)s.%(ext)s" "视频URL"

# 按上传者分类保存
yt-dlp -o "~/Videos/%(uploader)s/%(title)s.%(ext)s" "视频URL"

# 限制下载速度为 2MB/s
yt-dlp --limit-rate 2M "视频URL"

# 使用代理
yt-dlp --proxy socks5://127.0.0.1:1080 "视频URL"

# 查看所有可用格式
yt-dlp -F "视频URL"

# 下载特定格式 ID
yt-dlp -f 137 "视频URL"

# 只获取视频信息不下载
yt-dlp --skip-download --write-info-json "视频URL"

# 打印视频详细信息
yt-dlp --dump-json "视频URL"


## 常用组合命令

# 下载 YouTube 视频 + 中英文字幕
yt-dlp --write-subs --sub-langs zh-Hans,en "YouTube URL"

# 下载并转换为 MP4
yt-dlp --merge-output-format mp4 "视频URL"

# 下载 B 站视频最高画质（使用 cookies）
yt-dlp --cookies-browser chrome -f "best" "B站视频URL"

# 批量下载播放列表，跳过已下载
yt-dlp --download-archive downloaded.txt "播放列表URL"
